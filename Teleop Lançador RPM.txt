
package org.firstinspires.ftc.teamcode.teleOp;

import com.qualcomm.robotcore.eventloop.opmode.TeleOp;
import com.qualcomm.robotcore.eventloop.opmode.LinearOpMode;
import com.qualcomm.robotcore.hardware.DcMotor;
import com.qualcomm.robotcore.hardware.DcMotorEx;
import com.qualcomm.robotcore.util.Range;

@TeleOp(name = "TeleOp Lancador RPM", group = "Assistencias")
public class TeleOpLancadorRPM extends LinearOpMode {

    @Override
    public void runOpMode() {

        DcMotorEx MS = hardwareMap.get(
            DcMotorEx.class, "MS");

        MS.setZeroPowerBehavior(
            DcMotor.ZeroPowerBehavior.FLOAT);

        MS.setMode(DcMotor.RunMode.RUN_USING_ENCODER);

        final double MAX_RPM = 3000.0;

        double targetRPM = 0.0;
        double power = 0.0;

        waitForStart();

        while (opModeIsActive()) {

            // Leitura analogica do RT
            double trigger = gamepad1.right_trigger;

            // Converte a pressao em potencia
            power = Range.clip(trigger, 0.0, 1.0);

            // Calcula a RPM solicitada estimada
            targetRPM = power * MAX_RPM;

            // Aciona o lancador
            MS.setPower(power);

            // Desliga pelo botao Y
            if (gamepad1.y) {
                MS.setPower(0.0);
                power = 0.0;
                targetRPM = 0.0;
            }

            telemetry.addData(
                "Pressao RT", "%.2f", trigger);

            telemetry.addData(
                "RPM solicitada (estimada)", "%.0f", targetRPM);

            telemetry.addData(
                "Potencia aplicada", "%.2f", power);

            telemetry.addData(
                "Lancador ativo", MS.getPower() > 0);

            telemetry.addData(
                "Encoder", MS.getCurrentPosition());

            telemetry.update();
        }

        MS.setPower(0.0);
    }
}
